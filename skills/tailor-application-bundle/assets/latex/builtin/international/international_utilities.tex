\newcommand{\resumeHeader}[4]{%
  \begin{center}
    {\Huge\scshape #1}\\[-1pt]
    \if\relax\detokenize{#2}\relax\else{\large #2}\\[-1pt]\fi
    \small
    \if\relax\detokenize{#3}\relax\else\faMapMarker*\ #3\fi
    \if\relax\detokenize{#4}\relax\else
      \if\relax\detokenize{#3}\relax\else\quad\fi #4
    \fi
  \end{center}\vspace{-10pt}
}

  \noindent\begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}
    \textbf{#1} & \textbf{\small #2}\\
    \textit{\small #3} & \textit{\small #4}
  \end{tabular*}\vspace{-2pt}
}

}
